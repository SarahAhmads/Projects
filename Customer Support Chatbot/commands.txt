=== SETUP (FIRST TIME ONLY) ===
# The processed data folder is stored as a ZIP to keep the repo lightweight.
# You MUST extract it before running the project.

# Windows (PowerShell):
Expand-Archive -Path "data\processed.zip" -DestinationPath "data\" -Force
# Linux / macOS:
unzip data/processed.zip -d data/

# After extraction, you should have the folder: data/processed/


=== BACKEND (FASTAPI) ===
# Run these commands in the project root directory

# 1. Create a Python virtual environment (if not already done)
python -m venv venv

# 2. Activate the virtual environment
# Windows (PowerShell):
.\venv\Scripts\Activate.ps1
# Linux / macOS:
source venv/bin/activate

# 3. Install Python dependencies
pip install --default-timeout=1000 -r requirements.txt

# 4. Configure .env file in the root directory with:
# OPENROUTER_API_KEY=your_openrouter_api_key_here
# API_KEY=dev-secret-key

# 5. Start the backend server
uvicorn backend.api.main:app --host 127.0.0.1 --port 8001 --reload


=== FRONTEND (REACT + VITE) ===
# Open a new, separate terminal window and run:

# 1. Navigate to the frontend folder
cd frontend

# 2. Install node dependencies
npm install

# 3. Start the Vite development server
npm run dev
